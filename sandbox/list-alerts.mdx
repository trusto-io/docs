---
title: "list-alerts"
sidebarTitle: "List alerts"
description: "Interactively retrieve and filter Trusto alerts."
api: "GET /api/v1/alerts"
authMethod: "key"
playground: "interactive"
---

Retrieve a paginated collection of alerts available to the organization associated with your API key. The sandbox environment is selected by default; use the server selector to switch environments. Array-valued filters accept comma-separated values.

<ParamField query="page" default="0" type="integer">
  Zero-based page number.
</ParamField>

<ParamField query="size" default="10" type="integer">
  Number of alerts per page.
</ParamField>

<ParamField query="q" type="string">
  Search provider ID, merchant descriptor, card BIN, or card last four.
</ParamField>

<ParamField query="startDate" type="string">
  Minimum alert date in `YYYY-MM-DD` format.
</ParamField>

<ParamField query="endDate" type="string">
  Maximum alert date in `YYYY-MM-DD` format.
</ParamField>

<ParamField query="transactionStartDate" type="string">
  Minimum transaction date in `YYYY-MM-DD` format.
</ParamField>

<ParamField query="transactionEndDate" type="string">
  Maximum transaction date in `YYYY-MM-DD` format.
</ParamField>

<ParamField query="provider" type="string">
  Comma-separated providers: `ETHOCA`, `VERIFI`, or `CDRN`.
</ParamField>

<ParamField query="status" type="string">
  Comma-separated statuses: `OPEN`, `RESOLVED`, `REFUNDED`, `CHARGEBACK`, `EXPIRED`, or `INVALID`.
</ParamField>

<ParamField query="expired" type="boolean">
  Filter by whether the alert has expired.
</ParamField>

<ParamField query="expiresIn" type="integer">
  Return unresolved alerts expiring within this many hours.
</ParamField>

<ParamField query="outcome" type="string">
  Comma-separated stored outcome values.
</ParamField>

<ParamField query="descriptor" type="string">
  Filter by merchant descriptor prefix.
</ParamField>

<ParamField query="type" type="string">
  Filter by an exact alert type.
</ParamField>

<ParamField query="companyId" type="string">
  Comma-separated company UUIDs.
</ParamField>

<ParamField query="sortField" type="string">
  Alert property used for sorting.
</ParamField>

<ParamField query="sortOrder" default="ASC" type="string">
  Sort direction: `ASC` or `DESC`.
</ParamField>

<ResponseField name="data" type="Alert[]" required>
  Alerts for the current page, including `outcome` and `hoursLeft`.
</ResponseField>

<ResponseField name="total" type="integer" required>
  Total matching alerts.
</ResponseField>

<ResponseField name="totalPages" type="integer" required>
  Total pages.
</ResponseField>

<ResponseField name="page" type="integer" required>
  Current zero-based page.
</ResponseField>

<ResponseField name="size" type="integer" required>
  Requested page size.
</ResponseField>
